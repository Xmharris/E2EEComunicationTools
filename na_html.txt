		function findPlace(results, name){
			var result = null;
			results.forEach((item) => {
				if(item.types[0] == name && item.types[1] == "political") {
					result = name == 'country' ? item.long_name : item.short_name;
				}
			});
			return result;

		var map, markers = [], lastinfowindow, geocoder = new google.maps.Geocoder();

		function useMyLocation(btnUsed) {
			$('.loading-phonelines').show();
			$('.phoneline-error').html("").hide();
			navigator.geolocation.getCurrentPosition(
				(position) => {
					const latlng = {
						lat: parseFloat(position.coords.latitude),
						lng: parseFloat(position.coords.longitude)
					};

					geocoder
						.geocode({ location: latlng })
						.then((response) => {
							var city = findPlace(response.results[0].address_components, 'locality'),
							state = findPlace(response.results[0].address_components, 'administrative_area_level_1'),
							country = findPlace(response.results[0].address_components, 'country');

							$('#txtSearchArea').val('' /*`${city}, ${state}, ${country}`*/);
							$('#txtSearchArea').attr('data-lat', latlng.lat);
							$('#txtSearchArea').attr('data-lng', latlng.lng);
							$('#txtSearchArea').attr('data-state', state);
							$('#txtSearchArea').attr('data-country', country);

							loadData(latlng.lat, latlng.lng, 0, country, state);
							$('.loading-phonelines').hide();
						})
						.catch((e) => {
							console.log("Geocoder failed due to: " + e);
							$('.loading-phonelines').hide();
						});
				}, (error) => {
					console.log(error);
					$('.loading-phonelines').hide();
					if(btnUsed) {
						switch(error.code) {
							case error.PERMISSION_DENIED:
								$('.phoneline-error').html("An error occurred while accessing your location. It seems that location services are disabled. Enter the name of a city, or enable location services in your browser.").show();
								break;
							case error.POSITION_UNAVAILABLE:
								$('.phoneline-error').html("An error occurred while accessing user location. Location information is unavailable.").show();
								break;
							case error.TIMEOUT:
								$('.phoneline-error').html("An error occurred while accessing user location. The request to get user location timed out.").show();
								break;
							case error.UNKNOWN_ERROR:
								$('.phoneline-error').html("An error occurred while accessing user location.").show();
								break;
						}
					}
				},{
					enableHighAccuracy: true,
					timeout: 10000,
				}
			);

		function searchLocations()
		{
			var search_term = $('#txtSearchArea').val();
			if(search_term.length > 2) {
				geocoder.geocode( { 'address': search_term}, function(results, status) {
					if (status == 'OK') {
						$('.places').html('');
						//console.log(results);
						results.forEach((item) => {
							var p_country = '', p_state = '';
							if(item.address_components !== undefined && item.address_components.length > 0) {
								item.address_components.forEach(item => {
									if(item.types.includes("administrative_area_level_1")) {
										p_state = item.long_name;
									} else if(item.types.includes("country")) {
										p_country = item.long_name;
									}

		function placeMarker(marker_img, marker_lat, marker_long, marker_content, mapObj, marker_id, marker_arr) {
            var mylatlng = new google.maps.LatLng(marker_lat, marker_long);
            var infowindow = new google.maps.InfoWindow({
                content: "<div style='height: 120px; width: 280px;'>" + marker_content + "</div>"
            });
            var marker = new google.maps.Marker({
                map: mapObj,
				icon: marker_img,
                position: mylatlng
            });
            google.maps.event.addListener(marker, "click", function() {
                if(lastinfowindow instanceof google.maps.InfoWindow) lastinfowindow.close();
                infowindow.open(map, this);
                lastinfowindow = this.infowindow;
            });
            marker.orderid = marker_id;
            marker.infowindow = infowindow;
            marker_arr[marker_arr.length] = marker;
        }

		function cleanPhoneNumber(str) {
			str = $.trim(str);
			var sanitized = str.replace(' ', '').replace('-', '').replace('(', '').replace(')', '').replace('.', '');

			return `<div class="phne"><a href="tel:${sanitized}">${str}</a></div>`;

		}

		function disaplyDistance(country, distance) {
			var str = '';
			if(country === 'United States' || country === 'United Kingdom') {
				str += `<div class="distance">${Number.parseFloat(distance).toFixed(2)} miles</div>`;
			} else {
				str += `<div class="distance">${Number.parseFloat(distance * 1.60934).toFixed(2)} kms</div>`;
			}
			return str;
		}

		function loadData(lat, lng, within, country, state) {

			var action = 'search';

			$.post( "https://na.org/wp-content/plugins/meetings-finder/ajax.php",
            {
                action, lat, lng, within, country, state
            })
            .done(function( data ) {
				var jsonData = JSON.parse(data);
				if(jsonData.status == 'success') {
					var latlngC = new google.maps.LatLng(lat, lng);

					var zoom = 5;

					// dynamic zoom
					var myOptions = {
						zoom: zoom,
						center: latlngC,
						mapTypeId: google.maps.MapTypeId.ROADMAP,
						scaleControl: true,
						scaleControlOptions: { position: google.maps.ControlPosition.RIGHT_TOP }
					};
					map = new google.maps.Map(document.getElementById("hr-map"),myOptions);

					var circle = null, radius = jsonData.radius * 1609.34;
					if (circle) circle.setMap(null);
					circle = new google.maps.Circle({
						center: latlngC,
						radius: radius,
						strokeColor: "#1978C1",
						strokeOpacity: 0.8,
						strokeWeight: 1,
						fillColor: "#1978C1",
						fillOpacity: 0.15,
						map: map
					});
					map.fitBounds(circle.getBounds());

					$('.phonelines-results .list').html('');
					var counter = 0;
					jsonData.data.forEach((item) => {
						var ts = new Date();
						var li = '';
						if($.trim(item.description).length > 0) {
							if(item.phonewebid.length > 0) {
								li += `<div class="description"><a href="https://na.org/wp-content/plugins/meetings-finder/orphans.php?id=${item.phonewebid}&#038;ts=${ts.getTime()}" class="orphan">${$.trim(item.description)}</a></div>`;
							} else if($.trim(item.website).length > 0) {
								var web = $.trim(item.website);
								if(web.substr(0, 4) != 'http') {
									web = 'http://' + web;
								}
								li += `<div class="description"><a href="${web}" title="${web}" class="external-link" target="_blank">${$.trim(item.description)}</a></div>`;
							} else {
								li += `<div class="description">${$.trim(item.description)}</div>`;
							}
						}
						var csp = '';
						if($.trim(item.city).length > 0) { csp = $.trim(item.city); }
						if($.trim(item.state).length > 0) { csp += csp.length > 0 ? ', ' : ''; csp += $.trim(item.state); }
						if($.trim(item.postal_code).length > 0) { csp += csp.length > 0 ? ' - ' : ''; csp += $.trim(item.postal_code); }
						if(csp.length > 0) {
							li += `<div class="csp">${csp}${item.country !== 'United States' ? ',' : ''}${disaplyDistance(item.country, item.distance)}</div>`;
						}
						if($.trim(item.country).length > 0 && item.country !== 'United States') {
							li += `<div class="country">${$.trim(item.country)}${csp.length == 0 ? disaplyDistance(item.country, item.distance) : ''}</div>`;
						}
						if($.trim(item.phone).length > 0) {
							li += cleanPhoneNumber(item.phone);
						}
						if($.trim(item.phone2).length > 0) {
							li += cleanPhoneNumber(item.phone2);
						}
						if($.trim(item.whatsapp).length > 0) {
							li += `<div class="whatsapp">WhatsApp: ${$.trim(item.whatsapp)}</div>`;
						}

						$('.phonelines-results .list').append(`<div class="item">${li}</div>`);

						var marker_img = 'https://na.org/wp-content/plugins/meetings-finder/img/NAMarkerOther32.png';
						if(item.type == 'Region') {
							marker_img = 'https://na.org/wp-content/plugins/meetings-finder/img/NAMarkerRegion32.png';
						}

						placeMarker(marker_img, item.latitude, item.longitude, li, map, counter++, markers);
					});

					/*

					}

					var ll = '', us_ca = country == 'United States' || country == 'Canada';

					ll += `<div class="tab_content us_ca" ${us_ca ? '' : 'style="display: none;"'}>`;
